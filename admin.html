const ADMIN_PASS = "ADMIN_SECRET_PASS_2026"; const BACKEND_URL = window.location.origin; function updateAdminDashboard() { fetch(`${BACKEND_URL}/api/admin/requests`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ adminPassword: ADMIN_PASS }) }) .then(res => res.json()) .then(data => { if (data.success) { document.getElementById('total-users-badge').innerText = data.totalUsers; // يتم تعبئة الجداول هنا تلقائياً بناءً على مصفوفة requests و allUsers } }); } setInterval(updateAdminDashboard, 4000);
