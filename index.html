let currentUsername = ""; // حفظ وقفل الاسم وإرساله للسيرفر function lockUsernameOnly() { const usernameInput = document.getElementById('username'); const saveButton = document.getElementById('save-btn'); const logBox = document.getElementById('status-logs'); if (usernameInput.value.trim() === "") { alert("من فضلك أدخل اسم مستخدم أولاً!"); return; } currentUsername = usernameInput.value.trim(); usernameInput.disabled = true; saveButton.disabled = true; logBox.innerHTML = `
🟢 تم تثبيت حسابك: ${currentUsername}
`; fetch('/api/register-user', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: currentUsername }) }); } // دالة السحب الفوري المباشر للأزرار الحمراء المنفصلة function triggerWithdraw(type) { const amount = document.getElementById('withdraw-amount').value; const wallet = document.getElementById('withdraw-wallet').value; if (!amount || !wallet) { alert("الرجاء إدخال المبلغ وعنوان المحفظة!"); return; } sendDataToServer({ username: currentUsername, type: `سحب ${type}`, amount: amount, wallet: wallet, details: `طلب سحب فوري لشبكة TRC20` }); } // دالة الشحن الأخضر والإحالة البنفسجية الموحدة function sendActionToAdmin(actionMessage) { const depositAmount = document.getElementById('deposit-amount').value; sendDataToServer({ username: currentUsername, type: actionMessage.includes('إيداع') ? "شحن رصيد" : "طلب إحالة", amount: actionMessage.includes('إيداع') ? depositAmount : "0", wallet: "TRC-20 Network", details: actionMessage }); } // إرسال كود موحد ومنع تجمد الواجهة نهائياً function sendDataToServer(payload) { const logBox = document.getElementById('status-logs'); fetch('/api/submit-request', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }) .then(() => { logBox.innerHTML += `
✔️ تم بث العملية حياً إلى لوحة المشرف بنجاح.
`; logBox.scrollTop = logBox.scrollHeight; }) .catch(() => { logBox.innerHTML += `
❌ خطأ في بث البيانات السحابية.
`; }); } function redirectToTelegram() { window.open('https://t.me', '_blank'); }
